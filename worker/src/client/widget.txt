/*!
 * SMS Verify Widget — 手机号验证弹窗
 *
 * 用法（落地页只需三行）：
 *   <script>window.SMS_VERIFY_CONFIG = { apiBase: 'https://xxx.workers.dev', site: 'ai-lander' };</script>
 *   <script src="https://xxx.workers.dev/w.js" async></script>

 * 脚本会自动接管页面里的 gotolink(urls)：
 *   点击 → 弹窗 → 人机检测 → 发验证码 → 校验通过 → 跳转 WhatsApp

  function emit(name, detail) {
    try {
      window.dispatchEvent(new CustomEvent(name, { detail: detail }));
      if (typeof CFG.onEvent === 'function') CFG.onEvent(name, detail);
    } catch (e) { /* ignore */ }

  function post(path, body) {
    return fetch(API + path, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify(body || {}),
    }).then(function (r) {
      return r.json().catch(function () { return {}; });
    });
  }

  function grant() {
    try { return JSON.parse(localStorage.getItem(STORE_KEY) || 'null'); } catch (e) { return null; }
  }

  function saveGrant(data) {
    if (!data || !data.token) return;
    try {
      localStorage.setItem(STORE_KEY, JSON.stringify({
        token: data.token,
        exp: data.expiresAt || null,
        phone: data.phone || null,
      }));
    } catch (e) { /* ignore */ }
  }

  /** 与页面原有 gotolink 一致的链接选择逻辑（优先复用页面的 localStorage 缓存） */
  function resolveUrl() {
    var urls = state.urls || [];
    if (!urls.length) return null;
    var key = 'link1';
    var cached = null;
    try {
      if (typeof window.getLocalStorageWithExpiry === 'function') {
        cached = window.getLocalStorageWithExpiry(key);
      } else {
        var raw = JSON.parse(localStorage.getItem(key) || 'null');
        if (raw && raw.expiry > Date.now()) cached = raw.value;
      }
    } catch (e) { /* ignore */ }
    if (cached) return cached;

    var pick = urls[Math.floor(Math.random() * urls.length)];
    try {
      var ttlSeconds = typeof window.expTime === 'number' ? window.expTime : 0;
      var ttl = ttlSeconds > 0 ? ttlSeconds * 1000 : 60 * 1000; // 页面未设置有效期时按 1 分钟缓存
      if (typeof window.setLocalStorageWithExpiry === 'function') {
        window.setLocalStorageWithExpiry(key, pick, ttl);
      } else {
        localStorage.setItem(key, JSON.stringify({ value: pick, expiry: Date.now() + ttl }));
      }
    } catch (e) { /* ignore */ }
    return pick;
  }

    var h = String(hex || '#30a05c').replace('#', '');
    if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2];
    var n = parseInt(h, 16);
    return 'rgba(' + ((n >> 16) & 255) + ',' + ((n >> 8) & 255) + ',' + (n & 255) + ',' + alpha + ')';

  function fillCountries() {
    var list = (state.remote && state.remote.countries) || [];
    el.country.innerHTML = '';
    list.forEach(function (c) {
      var opt = document.createElement('option');
      opt.value = c.iso2;
      opt.textContent = c.iso2 + ' +' + c.dial;
      opt.dataset.example = c.example || '';
      el.country.appendChild(opt);
    });
    var preferred = CFG.defaultCountry || (state.remote && state.remote.defaultCountry);
    if (preferred) el.country.value = preferred;
    updatePlaceholder();
  }

  function updatePlaceholder() {
    var opt = el.country.options[el.country.selectedIndex];
    el.phone.placeholder = (opt && opt.dataset.example) || 'phone number';
  }

  function showError(node, msg) {
    if (!msg) { node.classList.remove('sv-show'); node.textContent = ''; return; }
    node.textContent = msg;
    node.classList.add('sv-show');
  }

  function busy(btn, on, labelBusy, labelIdle) {
    btn.disabled = on;
    btn.innerHTML = on ? '<span class="sv-spin"></span>' + labelBusy : labelIdle;
  }

  /* ───────────── Turnstile ───────────── */

  function loadTurnstile() {
    return new Promise(function (resolve) {
      if (window.turnstile) return resolve(true);
      if (document.getElementById('sv-ts-script')) {
        var t = setInterval(function () {
          if (window.turnstile) { clearInterval(t); resolve(true); }
        }, 120);
        setTimeout(function () { clearInterval(t); resolve(Boolean(window.turnstile)); }, 8000);
        return;
      }
      var s = document.createElement('script');
      s.id = 'sv-ts-script';
      s.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js?render=explicit';
      s.async = true;
      s.onload = function () { resolve(Boolean(window.turnstile)); };
      s.onerror = function () { resolve(false); };
      document.head.appendChild(s);
    });
  }

  function renderTurnstile() {
    var rc = state.remote;
    if (!rc || !rc.turnstileEnabled || !rc.turnstileSiteKey || !window.turnstile) return;
    try {
      if (state.tsWidgetId !== null) window.turnstile.remove(state.tsWidgetId);
      state.tsToken = null;
      state.tsWidgetId = window.turnstile.render(el.tsBox, {
        sitekey: rc.turnstileSiteKey,
        theme: 'dark',
        size: 'flexible',
        callback: function (token) { state.tsToken = token; },
        'expired-callback': function () { state.tsToken = null; },
        'error-callback': function () { state.tsToken = null; },
      });
    } catch (e) { /* ignore */ }
  }

  /* ───────────── 流程 ───────────── */

  function showStep(step) {
    state.step = step;
    el.phoneBody.classList.toggle('sv-hide', step !== 'phone');
    el.codeBody.classList.toggle('sv-hide', step !== 'code');
    el.okBody.classList.toggle('sv-hide', step !== 'ok');
    if (step === 'code') setTimeout(function () { el.code.focus(); }, 60);
    if (step === 'phone') setTimeout(function () { el.phone.focus(); }, 60);
  }

  function open(urls) {
    state.urls = urls && urls.length ? urls : null;
    mount();
    applyTexts();
    fillCountries();
    showError(el.errPhone, '');
    showError(el.errCode, '');
    if (!(state.remote && state.remote.countries && state.remote.countries.length)) {
      showError(el.errPhone, text('errorCountry'));
    }
    showStep('phone');
    el.root.classList.add('sv-on');
    emit('sms:modal_open', {});
    setTimeout(function () { el.phone.focus(); }, 80);
    if (state.remote && state.remote.turnstileEnabled) {
      loadTurnstile().then(function (ok) { if (ok) renderTurnstile(); });
    }
  }

  function close() {
    if (el.root) el.root.classList.remove('sv-on');
    if (state.resendTimer) { clearInterval(state.resendTimer); state.resendTimer = null; }
  }

  /** 点击入口：先看本地凭证，有效则直接跳转（后台异步复核） */
  function handleClick(urls) {
    var g = grant();
    if (!g || !g.token) {
      ensureReady().then(function (rc) {
        if (!rc && FAIL_OPEN) { jump(urls); return; } // 接口不可用时放行，避免丢转化
        open(urls);
      });
      return;
    }

    // 预开窗口以保住浏览器手势，避免被拦截
    var win = null;
    try { win = window.open('', '_blank'); } catch (e) { win = null; }

    post('/api/grant', { token: g.token }).then(function (res) {
      if (res && res.ok && res.valid) {
        var url = res.redirectUrl || resolveUrlFor(urls);
        emit('sms:grant_reused', { phone: res.phone });
        navigate(win, url);
      } else {
        if (win) { try { win.close(); } catch (e) { /* ignore */ } }
        try { localStorage.removeItem(STORE_KEY); } catch (e) { /* ignore */ }
        ensureReady().then(function (rc) {
          if (!rc && FAIL_OPEN) { jump(urls); return; }
          open(urls);
        });
      }
    }).catch(function () {
      if (FAIL_OPEN) navigate(win, resolveUrlFor(urls));
      else {
        if (win) { try { win.close(); } catch (e) { /* ignore */ } }
        ensureReady().then(function () { open(urls); });
      }
    });
  }

  function resolveUrlFor(urls) {
    var prev = state.urls;
    state.urls = urls;
    var u = resolveUrl();
    state.urls = prev;
    return u;
  }

  function navigate(win, url) {
    if (!url) { if (win) { try { win.close(); } catch (e) {} } alert('No link available'); return; }
    if (win && !win.closed) win.location.href = url;
    else window.location.href = url;
    emit('sms:redirect', { url: url });
  }

  function jump(urls) {
    var url = resolveUrlFor(urls);
    if (url) window.location.href = url;
  }

  function ensureReady() {
    if (state.remote) return Promise.resolve(state.remote);
    if (state.remoteError) return Promise.resolve(null);
    return fetch(API + '/api/config')
      .then(function (r) { return r.json(); })
      .then(function (res) {
        state.remote = (res && res.config) || {};
        return state.remote;
      })
      .catch(function () {
        state.remoteError = true;
        if (CFG.fallbackConfig) state.remote = CFG.fallbackConfig;
        return state.remote || null;
      });
  }

  function onSend() {
    if (state.busy) return;
    showError(el.errPhone, '');

    var rc = state.remote || {};
    if (rc.turnstileEnabled && rc.turnstileSiteKey && !state.tsToken) {
      // 页面刚打开就点的情况：先把 Turnstile 渲染出来
      loadTurnstile().then(function (ok) {
        if (ok) renderTurnstile();
        showError(el.errPhone, text('errorTurnstile'));
      });
      return;
    }

    state.busy = true;
    busy(el.send, true, text('sending'), text('sendBtn'));

    post('/api/send', {
      country: el.country.value,
      national: el.phone.value,
      turnstileToken: state.tsToken,
      site: SITE,
      locale: CFG.locale || 'en',
    }).then(function (res) {
      state.busy = false;
      busy(el.send, false, text('sending'), text('sendBtn'));

      if (res && res.ok) {
        state.id = res.id;
        state.phone = res.phone || el.phone.value;
        showStep('code');
        el.code.value = '';
        startResend(res.resendAfter || rc.resendIntervalSec || 60);
        emit('sms:sent', { id: res.id, provider: res.provider });
        return;
      }
      var key = ERROR_TEXT[res && res.error] || 'errorGeneric';
      showError(el.errPhone, text(key));
      if (state.tsWidgetId !== null) { try { window.turnstile.reset(state.tsWidgetId); } catch (e) {} state.tsToken = null; }
      emit('sms:error', { phase: 'send', error: res && res.error });
    }).catch(function () {
      state.busy = false;
      busy(el.send, false, text('sending'), text('sendBtn'));
      showError(el.errPhone, text('errorGeneric'));
      emit('sms:error', { phase: 'send', error: 'network' });
    });
  }

  function startResend(seconds) {
    var left = Number(seconds) || 0;
    if (state.resendTimer) clearInterval(state.resendTimer);
    if (left <= 0) { el.resend.disabled = false; el.resend.textContent = text('resend'); return; }
    el.resend.disabled = true;
    el.resend.textContent = text('resendIn', { s: left });
    state.resendTimer = setInterval(function () {
      left -= 1;
      if (left <= 0) {
        clearInterval(state.resendTimer);
        state.resendTimer = null;
        el.resend.disabled = false;
        el.resend.textContent = text('resend');
      } else {
        el.resend.textContent = text('resendIn', { s: left });
      }
    }, 1000);
  }

  function onResend() {
    if (state.busy || el.resend.disabled) return;
    state.busy = true;
    showError(el.errCode, '');
    post('/api/send', {
      country: el.country.value,
      national: el.phone.value,
      turnstileToken: state.tsToken,
      site: SITE,
      locale: CFG.locale || 'en',
    }).then(function (res) {
      state.busy = false;
      if (res && res.ok) {
        state.id = res.id;
        startResend(res.resendAfter || 60);
        emit('sms:sent', { id: res.id, provider: res.provider, resend: true });
      } else {
        var key = ERROR_TEXT[res && res.error] || 'errorGeneric';
        showError(el.errCode, text(key));
      }
    }).catch(function () {
      state.busy = false;
      showError(el.errCode, text('errorGeneric'));
    });
  }

  function onVerify() {
    if (state.busy) return;
    var code = (el.code.value || '').replace(/\D/g, '');
    if (code.length < 4) { showError(el.errCode, text('errorCode')); return; }
    showError(el.errCode, '');

    // 预开窗口保住手势
    var win = null;
    try { win = window.open('', '_blank'); } catch (e) { win = null; }

    state.busy = true;
    busy(el.verify, true, text('verifying'), text('verifyBtn'));

    post('/api/check', { id: state.id, code: code }).then(function (res) {
      state.busy = false;
      busy(el.verify, false, text('verifying'), text('verifyBtn'));

      if (res && res.ok) {
        saveGrant({ token: res.token, expiresAt: res.expiresAt, phone: state.phone });
        showStep('ok');
        emit('sms:verified', { id: state.id, phone: state.phone });
        var url = res.redirectUrl || resolveUrlFor(state.urls);
        if (state.id) post('/api/redirect', { id: state.id });
        setTimeout(function () { navigate(win, url); }, 550);
        return;
      }
      if (win) { try { win.close(); } catch (e) { /* ignore */ } }
      var key = ERROR_TEXT[res && res.error] || 'errorCode';
      showError(el.errCode, text(key));
      el.code.focus();
      emit('sms:error', { phase: 'check', error: res && res.error });
    }).catch(function () {
      state.busy = false;
      busy(el.verify, false, text('verifying'), text('verifyBtn'));
      if (FAIL_OPEN) { navigate(win, resolveUrlFor(state.urls)); return; }
      if (win) { try { win.close(); } catch (e) { /* ignore */ } }
      showError(el.errCode, text('errorGeneric'));
    });
  }

  /* ───────────── 接管 gotolink ───────────── */

  function installHook() {
    if (window.__smsVerifyInstalled) return;
    if (typeof window.gotolink !== 'function') { setTimeout(installHook, 50); return; }
    window.__smsVerifyInstalled = true;
    window.__smsVerifyOriginal = window.gotolink;
    window.gotolink = function (urls) {
      if (!urls || !urls.length) return window.__smsVerifyOriginal.apply(window, arguments);
      handleClick(Array.isArray(urls) ? urls : [urls]);
    };
  }

  window.SMSVerify = {
    open: function (urls) { ensureReady().then(function () { open(urls || (window.nbgoto || [])); }); },
    close: close,
    reset: function () { try { localStorage.removeItem(STORE_KEY); } catch (e) {} },
    getGrant: grant,
  };

  function boot() {
    // 未配置 apiBase 时完全不接管页面行为，保证引入脚本本身零风险
    if (!API) return;
    ensureReady();
    installHook();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', boot);
  } else {
    boot();
  }
})();
